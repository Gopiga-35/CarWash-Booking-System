CREATE TABLE CARWASH_TB (
    RECORDID       VARCHAR2(12 BYTE) PRIMARY KEY,
    CUSTOMERNAME   VARCHAR2(40 BYTE) NOT NULL,
    VEHICLENUMBER  VARCHAR2(20 BYTE) NOT NULL,
    WASHTYPE       VARCHAR2(20 BYTE) NOT NULL,
    BOOKING_DATE   DATE NOT NULL,
    TIMESLOT       VARCHAR2(20 BYTE),
    REMARKS        VARCHAR2(100 BYTE)
);

CREATE SEQUENCE CARWASH_SEQ
START WITH 10
INCREMENT BY 1
MAXVALUE 99;

SELECT CARWASH_SEQ.NEXTVAL FROM DUAL;
